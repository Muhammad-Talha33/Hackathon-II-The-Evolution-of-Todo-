{{- if eq .Values.service.type "NodePort" }}

   Access the frontend via NodePort:

      Node Port: {{ .Values.service.nodePort }}
      URL: http://localhost:{{ .Values.service.nodePort }}

   {{- if hasKey .Values "minikube" }}
   Or use Minikube service command:

      minikube service {{ include "frontend.fullname" . }} --url
      # Opens in browser automatically
   {{- end }}

{{- else if eq .Values.service.type "ClusterIP" }}

   The frontend is only accessible within the cluster.

   To access from your local machine, use port-forward:

      kubectl port-forward svc/{{ include "frontend.fullname" . }} 3000:{{ .Values.service.port }}

   Then access: http://localhost:3000

{{- else if eq .Values.service.type "LoadBalancer" }}

   Get the external IP:

      kubectl get svc {{ include "frontend.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}'

   Once available, access:
      - http://<EXTERNAL-IP>:{{ .Values.service.port }}

{{- end }}
